import os, re
errors = []
for f in os.listdir('..'):
    if f.startswith('sutra') and f.endswith('.vols.html'):
        with open(os.path.join('..', f), 'r', encoding='utf-8') as file:
            content = file.read()
        matches = re.findall(r'start=(\\d+).*?<span[^>]*class=\"[^\"]*page[^\"]*\"[^>]*>(\\d+)</span>', content, re.DOTALL)
        for s, p in matches:
            if s.isdigit() and p.isdigit() and int(s)%1000 != int(p):
                errors.append(f'{f}: start={s}, page={p}, 后三位={int(s)%1000:03d}')
with open('errors.txt', 'w') as out:
    if errors:
        out.write(f'找到{len(errors)}个错误:\\n\\n')
        for e in errors: out.write(e + '\\n')
        print(f'发现{len(errors)}个错误')
    else:
        out.write('✅ 所有一致\\n')
        print('✅ 所有一致')
